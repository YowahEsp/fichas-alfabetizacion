% ---- Fuentes de la familia Memima ------------------------------------------
\newfontface\ligada{MemimaM.ttf}[Path=./]
\newfontface\pautada{Mestra1M.ttf}[Path=./]
\newfontface\punteada{Mestra2M.ttf}[Path=./]

% ---- Parámetros (un solo sitio para cambiar tamaños) ------------------------
% Pauta = 0,275 em del cuerpo. 1 mm de pauta ≈ 10,34 pt de cuerpo.
\newlength{\cuerpopauta}  \setlength{\cuerpopauta}{62pt}   % pauta de 6 mm
\newlength{\huecopauta}   \setlength{\huecopauta}{12mm}    % aire entre renglones
\newlength{\huecobloque}  \setlength{\huecobloque}{9mm}    % aire entre bloques
\newlength{\altopicto}    \setlength{\altopicto}{19mm}     % alto del pictograma
\newcommand{\tammodelo}{\fontsize{38pt}{38pt}\selectfont}  % palabra modelo
\newcommand{\tamlectura}{\fontsize{30pt}{46pt}\selectfont}  % hojas de lectura
\newcommand{\maxrepasos}{3}                                % copias en el repaso

% ---- Pauta ----------------------------------------------------------------
\newcommand{\pautasize}{\fontsize{\cuerpopauta}{\cuerpopauta}\selectfont}
\newcommand{\ps}{\char32 }            % espacio que SÍ dibuja pauta
% v1.2: la pauta se dibuja de fondo a todo el ancho y el texto va encima, para
% que el renglón de repaso acabe en el mismo punto que los vacíos.
% Renglón de altura fija: la caja mide solo la pauta, así el texto (astas,
% colas) no desplaza el renglón y todos quedan a la misma distancia.
% #1 = fuente (\pautada o \punteada), #2 = texto
\newcommand{\renglon}[2]{%
  \par\nointerlineskip\vskip\huecopauta
  \noindent\hbox to\linewidth{#1\pautasize
    \vrule height 0.275\cuerpopauta depth 0pt width 0pt
    \rlap{\hbox to\linewidth{\xleaders\hbox{\char32}\hfill}}%  pauta de fondo a ancho completo
    \smash{#2}\hfill}\par}

% Renglón de repaso: repite la palabra tantas veces como quepa (máx. \maxrepasos)
\newcount\nrep \newcount\irep
\newcommand{\repaso}[1]{%
  \setbox0=\hbox{\punteada\pautasize #1\ps}%
  \nrep=\numexpr(\number\linewidth - \number\wd0/2)/\number\wd0\relax
  \ifnum\nrep>\maxrepasos\relax \nrep=\maxrepasos\relax\fi
  \ifnum\nrep<1 \nrep=1 \fi
  \renglon{\punteada}{\irep=0 \loop #1\ps\advance\irep by 1 \ifnum\irep<\nrep\repeat}}

% ---- Bloque de trabajo de la ficha de lecto-escritura (Tipo A) -----------------
% \bloque{pictograma}{palabra}  pictograma = nombre del PNG en pictos/ sin extensión
\newcommand{\bloque}[2]{%
  \par\noindent
  \parbox[c][\altopicto]{30mm}{\includegraphics[height=\altopicto,width=26mm,keepaspectratio]{pictos/#1}}%
  \hfill
  \parbox[c]{0.6\linewidth}{\raggedleft{\ligada\tammodelo #2}\hspace*{4mm}}\par
  \repaso{#2}%
  \renglon{\pautada}{}%
  \renglon{\pautada}{}%
  \par\vskip\huecobloque}

% ---- Hoja de lectura (Tipo B) ------------------------------------------------
% \lpalabras{p1, p2, p3, ...}  palabras sueltas en columnas
% \lfrase{texto}               una frase por línea
% \lparrafo{texto}             párrafo corrido
\newcommand{\lpalabras}[1]{%
  \par\noindent{\ligada\tamlectura\raggedright
  \foreach \w in {#1}{\w\hspace{1.6em plus 0.5em}}\par}\vskip 4mm}
% v1.4: el \par va dentro del grupo para que se aplique el interlineado de
% \tamlectura (46 pt); antes quedaba fuera y la distancia entre frases variaba
% según los rasgos descendentes de cada línea.
\newcommand{\lfrase}[1]{\par\noindent{\ligada\tamlectura #1\par}\vskip 6mm}
\newcommand{\lparrafo}[1]{\par\noindent{\ligada\tamlectura #1\par}\vskip 5mm}

% ---- Separador de fichas: cada ficha es una página -----------------------
\newcommand{\nuevaficha}{\clearpage}

% ---- Enunciado (libre, no validado) y atribución -------------------------
% \enunciado{pictograma de la acción}{texto para la docente}
\newcommand{\enunciado}[2]{%
  \par\noindent\raisebox{-0.35\height}{\includegraphics[height=11mm]{pictos/#1}}%
  \hspace{3mm}{\large #2}\par\vskip 4mm}
\newcommand{\atribucion}{%
  \vfill\noindent{\tiny\color{gray}Pictogramas: Sergio Palao. Origen: ARASAAC
  (https://arasaac.org). Licencia: CC BY-NC-SA. Propiedad: Gobierno de Aragón.}}
